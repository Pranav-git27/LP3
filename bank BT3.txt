pragma solidity ^0.8.0;

contract Bank{
    uint256 balance = 0;
    address public accOwner;

    constructor(){
        accOwner = msg.sender;
    }

    function Deposite() public payable {
        require(accOwner==msg.sender,"You are not the account owner!");
        require(msg.value > 0, "The amount should be greater than 0.");
        balance += msg.value;
    }

    function Withdraw() public payable {
        require(accOwner==msg.sender,"You are not the account owner!");
        require(msg.value > 0, "Withdrawal amount should be greater than 0.");
        balance -= msg.value;
    }

    function showBalance() public view returns(uint256){
        require(accOwner==msg.sender, "You are not the account owner!");
        return balance;
    }
}
